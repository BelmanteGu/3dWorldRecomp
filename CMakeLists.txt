  endif()

  message(FATAL_ERROR "WWHD_RENDERER must be BOTH, METAL or VULKAN")
endif()
set(WWHD_HAS_METAL OFF)
set(WWHD_HAS_VULKAN OFF)

    message(FATAL_ERROR "Metal is available only on macOS; select VULKAN")

  set(WWHD_HAS_METAL ON)
  enable_language(OBJCXX)

if(WWHD_RENDERER MATCHES "^(BOTH|VULKAN)$")
  set(WWHD_HAS_VULKAN ON)
endif()

# WWHD_BUNDLED_DEPS: build glslang, zlib, LZ4 (and SDL3 if not found) from pinned sources with the
# project's own compiler. Release builds use it so the prebuilt runtime and its dependencies all
# come from the toolchain the installer later links with (llvm-mingw, zig).
option(WWHD_BUNDLED_DEPS "build missing dependencies from pinned sources" OFF)
if(WIN32 AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC" OR WWHD_BUNDLED_DEPS)
  include(cmake/WindowsDependencies.cmake)
endif()
if(WIN32 AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
  # The MSVC ABI links no compiler runtime, but clang lowers 128-bit division to compiler-rt calls
  # (__udivti3; on ARM64 nothing in the MSVC libraries provides them): link clang's builtins.
  set(WWHD_CLANG_TARGET_ARG)
  if(CMAKE_CXX_COMPILER_TARGET)
    set(WWHD_CLANG_TARGET_ARG --target=${CMAKE_CXX_COMPILER_TARGET})
  endif()
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} ${WWHD_CLANG_TARGET_ARG} -print-libgcc-file-name -rtlib=compiler-rt
                  OUTPUT_VARIABLE WWHD_CLANG_BUILTINS OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(WWHD_CLANG_BUILTINS AND EXISTS "${WWHD_CLANG_BUILTINS}")
    link_libraries("${WWHD_CLANG_BUILTINS}")
  endif()
endif()

  wwhd_pipeline_runtime_identity("${CMAKE_SOURCE_DIR}" WWHD_PIPELINE_RUNTIME_ID)
  set_property(SOURCE runtime/src/gfx/vulkan/backend.cpp APPEND PROPERTY
    COMPILE_DEFINITIONS "WWHD_PIPELINE_RUNTIME_ID=\"${WWHD_PIPELINE_RUNTIME_ID}\"")
  # VK_NO_PROTOTYPES: the renderer loads every Vulkan function at run time (gfx/vulkan/loader.h), so the
  # executable starts without a Vulkan loader or with an older one, and can say what is missing
  target_compile_definitions(wwhd PRIVATE WWHD_HAS_VULKAN=1 VK_NO_PROTOTYPES)
  if(WWHD_HAS_METAL)
    # One app, two renderers: the Vulkan loader and glslang are weak imports, so the game still
    # starts (with Metal) on a Mac without them; the renderer checks for them before using Vulkan.
    # SYSTEM: searched after the project's own -I paths, so a Homebrew boost/fmt in the same prefix
    # (/opt/homebrew/include) can't replace the vendored ones (cemu_latte is built with those; a
    # different boost changed LatteDecompilerShader's layout and crashed the first shader translation)
    target_include_directories(wwhd SYSTEM PRIVATE $<TARGET_PROPERTY:Vulkan::Vulkan,INTERFACE_INCLUDE_DIRECTORIES>
                                            $<TARGET_PROPERTY:glslang::glslang,INTERFACE_INCLUDE_DIRECTORIES>)
    # Release builds (WWHD_BUNDLED_DEPS) compile glslang statically into the game and ship the Vulkan
    # loader and MoltenVK next to it (tools/release/macos_vulkan.py); only shared libraries are weak.
    target_link_options(wwhd PRIVATE "LINKER:-weak_library,$<TARGET_FILE:Vulkan::Vulkan>")
    get_target_property(WWHD_GLSLANG_TYPE glslang::glslang TYPE)
    if(WWHD_GLSLANG_TYPE STREQUAL "SHARED_LIBRARY")
      target_compile_definitions(wwhd PRIVATE WWHD_GLSLANG_WEAK=1)
      foreach(lib glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits)
        target_link_options(wwhd PRIVATE "LINKER:-weak_library,$<TARGET_FILE:${lib}>")
      endforeach()
    else()
      target_link_libraries(wwhd PRIVATE glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits)
    endif()
  elseif(APPLE)
    # SDL host on a Mac: Homebrew's loader is linked; SDL and the renderer find it in the process
    target_link_libraries(wwhd PRIVATE Vulkan::Vulkan glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits)
  else()
    # Windows, Linux, Android: SDL loads the Vulkan loader at run time (SDL_Vulkan_LoadLibrary);
    # only its headers are used here
    target_include_directories(wwhd SYSTEM PRIVATE $<TARGET_PROPERTY:Vulkan::Vulkan,INTERFACE_INCLUDE_DIRECTORIES>)
    target_link_libraries(wwhd PRIVATE glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits)
  endif()
endif()
